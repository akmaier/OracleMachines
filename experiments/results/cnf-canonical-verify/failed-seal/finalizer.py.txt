#!/usr/bin/env python3
"""Seal direct canonical recognition, reader equivalence and remaining publication costs."""
from pathlib import Path
import re,subprocess,sys
sys.path.insert(0,str(Path(__file__).resolve().parents[1]))
from experiments import compare_cnf_canonical_verify as study
base=study.base
POSTHOC=('experiments/validate_cnf_canonical_verify.py','experiments/profile_cnf_canonical_verify.py','experiments/finalize_cnf_canonical_verify.py')


def finalize(out):
    p=base.load(out/'protocol.json');parent=study.bound(p['parent']);v=base.load(out/'validation.json')
    for ledger in (parent['source_fingerprints'],p['source_fingerprints']):
        for f,h in ledger.items():assert base.sha(f)==h,f
    assert len(parent['source_fingerprints'])==613
    for path,entry in v['parent_manifests'].items():
        assert base.sha(path)==entry['sha256'];artifacts=base.load(path)['artifacts'];assert len(artifacts)==entry['artifacts_checked']
        for f,h in artifacts.items():assert base.sha(f)==h,f
    assert len(v['parent_manifests'])==114 and v['prior_artifacts_unchanged']==7859
    sources=dict(p['source_fingerprints']);sources.update({f:base.sha(f) for f in POSTHOC});assert len(sources)==619
    checkpoint=base.load(out/'checkpoint.json');assert checkpoint['reference_union']==parent['reference_union']
    checkpoint['source_fingerprints']=sources;base.write(out/'checkpoint.json',checkpoint)
    v['source_fingerprints']=sources;v['checkpoint']=study.binding(out/'checkpoint.json');base.write(out/'validation.json',v)
    assert sorted(v['preserved_tables'].values())==[22,26,34,42,53,28886]
    assert v['reference_union_unchanged'] and v['new_native']==v['new_extensions']==0
    assert (v['query_audits'],v['truth_tables'],v['choice_audits'],v['known_hit_audits'],v['exact_query_comparisons'])==(624,104,144,84,416)
    assert (v['old_composed_records'],v['old_pilot_records'])==(5650,444)
    assert (v['canonical_fixedpoints'],v['representation_mutations'])==(28886,4961)
    summaries=base.load(out/'report.json')['summary'];a=summaries['known'];b=summaries['linear']
    for k in ('native_digest','extension_digest','database_bytes','compressed_bytes','native_payload','extension_payload','cache_digest'):
        assert b[k]==a[k]
    assert b['counters']['canonical_form_reuses']==b['counters']['verified_nodes']==552
    assert b['counters']['canonical_form_literal_visits']==55601
    assert (a['counters']['normalization_literal_visits'],b['counters']['normalization_literal_visits'])==(175148,119547)
    assert a['counters']['key_serialized_bytes']==b['counters']['key_serialized_bytes']==678378
    attribution=base.load(out/'attribution.json')
    assert all(p[k]<0 for p in attribution['pairs'] if p['mode']=='linear' for k in ('query_seconds','cold_seconds','complete_worker_seconds'))
    profile=base.load(out/'profile.json')
    assert all(p['outputs_and_counters_identical'] for p in profile['profiles'].values())
    assert profile['profile_source']['sha256']==sources['experiments/profile_cnf_canonical_verify.py']
    c=profile['canonical_output_census']['_visit']
    assert (c['calls'],c['earlier_normalized_output_hits'],c['hit_literal_units'])==(546,45,8011)
    paper='paper/bounded_domain_sat_and_amortized_reasoning.tex';commit=subprocess.check_output(['git','rev-parse','94bf2be'],text=True).strip()
    old=subprocess.check_output(['git','show',commit+':'+paper],text=True);pattern=r'\\begin\{proof\}.*?\\end\{proof\}'
    proofs=re.findall(pattern,old,re.S);assert len(proofs)==64 and proofs==re.findall(pattern,Path(paper).read_text(),re.S)
    logs={name:(out/name).read_text() for name in ('full-tests.log','focused-tests.log','latex-final.log')}
    assert re.search(r'Ran 1271 tests in .*\n\nOK',logs['full-tests.log'])
    assert re.search(r'Ran 102 tests in .*\n\nOK',logs['focused-tests.log'])
    warnings={k:len(re.findall('ResourceWarning: unclosed database',x)) for k,x in logs.items() if k!='latex-final.log'}
    assert warnings=={'full-tests.log':44,'focused-tests.log':0}
    latex=logs['latex-final.log'];pages=int(re.search(r'\((\d+) pages,',latex)[1]);assert not re.search(r'Warning|Overfull|Underfull|^!',latex,re.M)
    checks=dict(passed=True,baseline_commit=commit,prior_sources=613,final_sources=len(sources),prior_artifacts=7859,prior_manifests=114,
        source_fingerprints=sources,full_tests=1271,focused_tests=102,warnings=warnings,proof_blocks_unchanged=64,manuscript_pages=pages,paper_sha256=base.sha(paper),
        limitations=['Direct recognition replaces sorting/renaming with explicit canonical-form checks; all key bytes and semantic proof counters remain unchanged.',
            'All 104 inputs replay the preceding study; no new SAT pattern coverage or general performance law from two orders.',
            'No retained canonical cache, but transient clauses/support sets, tuples and keys consume space.',
            'No new persisted proofs or answers; the previous archive and all decisions are referenced unchanged.',
            'Publication batching and remaining direct transport opportunities are next investigations, not implemented savings.',
            'No production-default change, new Lean theorem or unrestricted-SAT resource bound.'])
    base.write(out/'checks.json',checks)
    files=sorted(f for f in out.rglob('*') if f.is_file() and f.name!='manifest.json')
    base.write(out/'manifest.json',dict(artifacts={str(f):base.sha(f) for f in files}))
    print(base.encode(dict(passed=True,artifacts=len(files),next_artifacts=7859+len(files),next_manifests=115,final_sources=len(sources),pages=pages)),flush=True)


if __name__=='__main__':finalize(Path(sys.argv[1]))
