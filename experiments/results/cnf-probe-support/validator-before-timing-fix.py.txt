#!/usr/bin/env python3
"""Audit proof-support transport, frozen controls, native archives and lineage."""
from collections import Counter
from copy import deepcopy
from pathlib import Path
import gzip,re,subprocess,sys,tempfile,time
sys.path.insert(0,str(Path(__file__).resolve().parents[1]))
from experiments import compare_cnf_probe_support as study
base=study.base;old=study.old;previous=study.previous


def validate(out):
    start=time.perf_counter();p=base.load(out/'protocol.json');parent=old.bound(p['parent']);pv=base.load(study.PARENT/'validation.json')
    manifests=deepcopy(pv['parent_manifests']);mp=Path(p['parent_manifest']['path']);old.bound(p['parent_manifest'])
    manifests[str(mp)]=dict(sha256=base.sha(mp),artifacts_checked=len(base.load(mp)['artifacts']));artifacts=0
    for path,entry in manifests.items():
        assert base.sha(path)==entry['sha256']
        for f,h in base.load(path)['artifacts'].items():assert base.sha(f)==h,f;artifacts+=1
    assert (len(manifests),artifacts)==(182,23167)
    sources={**p['source_fingerprints'],'experiments/validate_cnf_probe_support.py':base.sha(__file__)}
    for f,h in sources.items():assert base.sha(f)==h,f
    cd=old.bound(p['cases']);cases=[old.bound(b) for b in cd['cases']];assert len(cases)==60 and len({c['root'] for c in cases})==60
    plans=[]
    for case in cases:
        c=Counter();frames,state=study.candidate.frozen.bind_prefix(case['source'],case['templates'],case['prefix'],c)
        plan=study.candidate.prepare_support(state['local'],[h['template'] for h in state['hints']],case['program'],c)
        if plan is not None:
            assert plan['source']==state['local']
            assert set(plan['indices'])<=set(range(len(state['local'])))
            assert study.native.normalize([state['local'][i] for i in plan['indices']])[0]==plan['small']
            study.candidate.frozen.validate(plan['small'],plan['templates'],plan['program'],Counter())
        plans.append(dict(name=case['name'],root=case['root'],binary_prefix_frames=len(frames),plan=plan,counts=dict(c)))
    study.prior.compact(out/'support-plans.json',dict(rows=plans))
    trials=base.load(out/'comparisons.json')['samples'];assert len(trials)==360;tg={};audits={};archives=[]
    for sample in trials:
        r=old.bound(sample);index=sample['index'];mode=sample['mode'];assert r['case']==cd['cases'][index] and r['mode']==mode
        assert r['result']['status']=='unsat' and r['result']['result']['root']==cases[index]['root']
        if mode in ('fresh','full'):
            origin=cd['origins'][index];oldmode='extension' if mode=='full' else 'fresh'
            earlier=base.load(Path(origin['directory'])/f'case-{origin["index"]}-{oldmode}-0/report.json')
            for key in ('result','counts','native_counts','authority_tables','digest','record_bytes'):assert r[key]==earlier[key],(index,mode,key)
        a=r['archive'];assert base.sha(a['path'])==a['sha256'];key=(a['sha256'],cases[index]['root'])
        if key not in audits:
            with tempfile.TemporaryDirectory() as td:
                path=Path(td)/'db';previous.restore_archive(a,path);stored,readers=previous.audit(path,[dict(source=cases[index]['source'])],[r['result']])
            assert {t:len(v) for t,v in stored.items()}==r['authority_tables']
            assert {t:base.key_sha(base.encode(sorted(v.items()))) for t,v in stored.items()}==r['digest']
            assert sum(len(k.encode())+len(v.encode()) for table in stored.values() for k,v in table.items())==r['record_bytes']
            audits[key]=dict(archive=a,root=cases[index]['root'],readers=readers);archives.append(a)
        tg.setdefault((index,mode),[]).append(r)
    totals={m:Counter() for m in study.MODES};counts={m:Counter() for m in study.MODES};rss=Counter();summary=[]
    for i,case in enumerate(cases):
        row=dict(name=case['name'],root=case['root'],snapshot_answer=case['snapshot_answer'],arms={})
        for mode in study.MODES:
            a,b=tg[(i,mode)]
            for key in ('result','counts','native_counts','digest','record_bytes','authority_tables'):assert a[key]==b[key],(i,mode,key)
            metrics={k:(a[k]+b[k])/2 for k in ('solve_seconds','warm_seconds','cold_seconds','persistence_seconds','complete_seconds','record_bytes')}
            metrics['records']=sum(a['authority_tables'].values());totals[mode].update(metrics);counts[mode].update(a['counts']);rss[mode]=max(rss[mode],a['peak_rss_bytes'],b['peak_rss_bytes'])
            row['arms'][mode]=metrics
        summary.append(row)
    comparisons={}
    for control in ('fresh','full'):
        comparisons[control]={k:dict(improved=sum(r['arms']['support'][k]<r['arms'][control][k] for r in summary),
            equal=sum(r['arms']['support'][k]==r['arms'][control][k] for r in summary),
            regressed=[r['name'] for r in summary if r['arms']['support'][k]>r['arms'][control][k]]) for k in ('solve_seconds','complete_seconds','record_bytes')}
    analysis=dict(cases=len(cases),totals=totals,counts=counts,peak_rss_bytes=rss,summary=summary,comparisons=comparisons,
        support_plans=dict(fallbacks=sum(r['plan'] is None for r in plans),smaller=sum(r['plan'] is not None and len(r['plan']['indices'])<len(r['plan']['source']) for r in plans),
            prefix_frames=sum(r['binary_prefix_frames'] for r in plans)),unique_archive_audits=len(audits),sources=len(sources),limitations=p['limitations'])
    base.write(out/'analysis.json',analysis)
    cp=deepcopy(parent);cp.update(parent_checkpoint=p['parent'],source_fingerprints=sources,
        probe_support={k:old.binding(out/(k+'.json')) for k in ('protocol','cases','comparisons','support-plans','analysis')})
    cp['probe_support']['supplemental_archives']=archives;base.write(out/'checkpoint.json',cp)
    base.write(out/'validation.json',dict(passed=True,parent_manifests=manifests,prior_artifacts_unchanged=artifacts,source_fingerprints=sources,
        reference_union_unchanged=cp['reference_union']==parent['reference_union'],preserved_tables=pv['preserved_tables'],new_cumulative_records=0,
        supplemental_archive_audits=list(audits.values()),seconds=time.perf_counter()-start,checkpoint=old.binding(out/'checkpoint.json')))
    print(base.encode(dict(totals=totals,comparisons=comparisons,plans=analysis['support_plans'],archives=len(audits))),flush=True)


def seal(out):
    v=base.load(out/'validation.json');assert v['passed'];cp=base.load(out/'checkpoint.json')
    for f,h in cp['source_fingerprints'].items():assert base.sha(f)==h,f
    paper='paper/bounded_domain_sat_and_amortized_reasoning.tex';before=subprocess.check_output(['git','show','9111e4b9:'+paper],text=True)
    pat=r'\\begin\{proof\}.*?\\end\{proof\}';proofs=re.findall(pat,before,re.S);assert len(proofs)==64 and proofs==re.findall(pat,Path(paper).read_text(),re.S)
    full=(out/'full-tests.log').read_text();focus=(out/'focused-tests.log').read_text();latex=(out/'latex-final.log').read_text()
    assert re.search(r'Ran 1888 tests in .*\n\nOK',full) and re.search(r'Ran 23 tests in .*\n\nOK',focus)
    assert not re.search(r'Warning|Overfull|Underfull|^!',latex,re.M)
    cp['probe_support']['decision']=old.binding(out/'decision.json');base.write(out/'checkpoint.json',cp)
    v['checkpoint']=old.binding(out/'checkpoint.json');base.write(out/'validation.json',v)
    base.write(out/'checks.json',dict(passed=True,full_tests=1888,focused_tests=23,proof_blocks_unchanged=64,manuscript_pages=int(re.search(r'\((\d+) pages,',latex)[1]),
        prior_sources=1050,final_sources=len(cp['source_fingerprints']),prior_manifests=182,prior_artifacts=23167,archive_audits=len(v['supplemental_archive_audits']),
        warnings=dict(full=full.count('ResourceWarning: unclosed database'),focused=focus.count('ResourceWarning: unclosed database')),checkpoint=old.binding(out/'checkpoint.json')))
    for f in out.rglob('*.log'):
        raw=f.read_bytes();Path(str(f)+'.raw.gz').write_bytes(gzip.compress(raw,mtime=0));f.write_text('\n'.join(line.rstrip() for line in raw.decode().splitlines()).rstrip()+'\n')
    files=sorted(f for f in out.rglob('*') if f.is_file() and f.name!='manifest.json')
    base.write(out/'manifest.json',dict(artifacts={str(f):base.sha(f) for f in files}))
    print(base.encode(dict(sealed=True,artifacts=len(files),next_manifests=183,next_artifacts=23167+len(files))))


if __name__=='__main__':globals()[sys.argv[1]](Path(sys.argv[2]))
